import {
  Injectable,
  BadRequestException,
  NotFoundException,
} from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';
import { ClockInDto } from './dto/clock-in.dto';
import { ClockOutDto } from './dto/clock-out.dto';
import { calculateHaversineDistance } from '../../common/utils/geo.util';
import { AttendanceStatus, LogType, GeofenceStatus } from '@prisma/client';

@Injectable()
export class AttendanceService {
  // Hardcoded as agreed. Can be moved to DB later.
  private readonly GEOFENCE_LAT = -6.904810338149059;
  private readonly GEOFENCE_LNG = 107.61031378245713;
  private readonly GEOFENCE_RADIUS_METERS = 50.0;
  //--6.904810338149059, 107.61031378245713
  constructor(private prisma: PrismaService) {}

  getGeofenceConfig() {
    return {
      geofence_id: 'GEO_RW07_SUKAMENAK',
      latitude: this.GEOFENCE_LAT,
      longitude: this.GEOFENCE_LNG,
      radius_meters: this.GEOFENCE_RADIUS_METERS,
    };
  }

  async clockIn(linmasId: string, dto: ClockInDto) {
    // 1. Double-check Geofence mathematically on backend
    const distance = calculateHaversineDistance(
      this.GEOFENCE_LAT,
      this.GEOFENCE_LNG,
      dto.client_gps.latitude,
      dto.client_gps.longitude,
    );

    if (distance > this.GEOFENCE_RADIUS_METERS) {
      throw new BadRequestException(
        `Geofence validation failed. You are ${Math.round(distance)}m away.`,
      );
    }

    // 2. Fetch the assignment and the associated file from B2
    const [assignment, photoFile] = await Promise.all([
      this.prisma.shiftAssignment.findUnique({
        where: { id: dto.shift_assignment_id },
        include: { shift: true },
      }),
      this.prisma.file.findFirst({ where: { cdn_url: dto.photo_url } }),
    ]);

    if (!assignment || assignment.linmasId !== linmasId) {
      throw new NotFoundException('Invalid shift assignment');
    }
    if (!photoFile) {
      throw new BadRequestException(
        'Photo file not found in system. Please upload first.',
      );
    }

    // 3. Determine if Late (e.g., > 15 mins after shift start)
    const now = new Date();
    const shiftStart = new Date(assignment.shift.startTime);
    // Adjust shift start to today for time comparison
    shiftStart.setFullYear(now.getFullYear(), now.getMonth(), now.getDate());

    const diffMins = (now.getTime() - shiftStart.getTime()) / 60000;
    const status =
      diffMins > 15 ? AttendanceStatus.LATE : AttendanceStatus.PRESENT;

    // 4. Save Session and Log
    const session = await this.prisma.attendanceSession.create({
      data: {
        linmasId: linmasId,
        shiftAssignmentId: assignment.id,
        status: status,
        logs: {
          create: {
            logType: LogType.CLOCK_IN,
            latitude: dto.client_gps.latitude,
            longitude: dto.client_gps.longitude,
            geofenceStatus: dto.geofence_status,
            photoFileId: photoFile.id,
          },
        },
      },
    });

    return {
      attendance_session_id: session.id,
      attendance_status: status,
      verified_distance_meters: parseFloat(distance.toFixed(2)),
    };
  }

  async clockOut(linmasId: string, dto: ClockOutDto) {
    const session = await this.prisma.attendanceSession.findFirst({
      where: { id: dto.attendance_session_id, linmasId: linmasId },
    });

    if (!session) throw new NotFoundException('Attendance session not found');

    const photoFile = await this.prisma.file.findFirst({
      where: { cdn_url: dto.photo_url },
    });

    if (!photoFile)
      throw new BadRequestException('Photo file not found in system.');

    const now = new Date(); // SERVER TIME

    // 1. Double-check Geofence mathematically
    const distance = calculateHaversineDistance(
      this.GEOFENCE_LAT,
      this.GEOFENCE_LNG,
      dto.client_gps.latitude,
      dto.client_gps.longitude,
    );

    let isInside = distance <= this.GEOFENCE_RADIUS_METERS;

    // 2. Log the Clock Out
    await this.prisma.attendanceLog.create({
      data: {
        sessionId: session.id,
        logType: LogType.CLOCK_OUT,
        latitude: dto.client_gps.latitude,
        longitude: dto.client_gps.longitude,
        geofenceStatus: isInside
          ? GeofenceStatus.VERIFIED_INSIDE
          : GeofenceStatus.VERIFIED_OUTSIDE,
        photoFileId: photoFile.id,
        timestamp: now, // TRUST THE SERVER TIME
      },
    });

    return {
      attendance_session_id: session.id,
      verified_distance_meters: parseFloat(distance.toFixed(2)),
      clock_out_time: now.toISOString(),
    };
  }

  async getRecap(month: number, year: number) {
    // Generate statistics for the Koordinator
    const sessions = await this.prisma.attendanceSession.findMany({
      where: {
        shiftAssignment: {
          shift: {
            shiftDate: {
              gte: new Date(year, month - 1, 1),
              lt: new Date(year, month, 1),
            },
          },
        },
      },
      include: {
        linmas: { include: { regu: true } },
      },
    });

    const recapMap = new Map();

    for (const session of sessions) {
      const lId = session.linmasId;
      if (!recapMap.has(lId)) {
        recapMap.set(lId, {
          linmas_id: lId,
          full_name: session.linmas.fullName,
          regu_name: session.linmas.regu?.name || null,
          present_count: 0,
          absent_count: 0,
          late_count: 0,
        });
      }

      const stats = recapMap.get(lId);
      if (session.status === AttendanceStatus.PRESENT) stats.present_count += 1;
      if (session.status === AttendanceStatus.LATE) stats.late_count += 1;
      if (session.status === AttendanceStatus.ABSENT) stats.absent_count += 1;
    }

    return Array.from(recapMap.values());
  }

  async getTodayAttendanceStatus(linmasId: string) {
    const now = new Date(); // SERVER TIME
    const startOfDay = new Date(
      now.getFullYear(),
      now.getMonth(),
      now.getDate(),
    );
    const endOfDay = new Date(
      now.getFullYear(),
      now.getMonth(),
      now.getDate() + 1,
    );

    // Find today's shift
    const assignment = await this.prisma.shiftAssignment.findFirst({
      where: {
        linmasId: linmasId,
        shift: { shiftDate: { gte: startOfDay, lt: endOfDay } },
      },
      include: { shift: true, attendanceSessions: true },
    });

    if (!assignment) {
      return {
        status: 'NO_SHIFT_TODAY',
        message: 'Anda tidak memiliki jadwal hari ini.',
      };
    }

    const shift = assignment.shift;
    const session = assignment.attendanceSessions[0];

    // Time calculations (Adjust database time to today's actual date)
    const shiftStartTime = new Date(shift.startTime);
    shiftStartTime.setFullYear(
      now.getFullYear(),
      now.getMonth(),
      now.getDate(),
    );

    const shiftEndTime = new Date(shift.endTime);
    shiftEndTime.setFullYear(now.getFullYear(), now.getMonth(), now.getDate());

    // Handle overnight shifts (e.g., 20:00 to 04:00)
    if (shiftEndTime < shiftStartTime) {
      shiftEndTime.setDate(shiftEndTime.getDate() + 1);
    }

    // Window: Can clock in 30 minutes before shift starts
    const earliestClockIn = new Date(shiftStartTime.getTime() - 30 * 60000);

    // Window: Can clock out up to 4 hours after shift ends
    const latestClockOut = new Date(shiftEndTime.getTime() + 4 * 60 * 60000);

    if (!session) {
      if (now < earliestClockIn) {
        const hh = shiftStartTime.getHours().toString().padStart(2, '0');
        const mm = shiftStartTime.getMinutes().toString().padStart(2, '0');
        return {
          status: 'WAITING',
          message: `Absen masuk dibuka pukul ${hh}:${mm}`,
        };
      }
      if (now > shiftEndTime)
        return { status: 'MISSED', message: 'Waktu shift sudah berakhir.' };
      return { status: 'CAN_CLOCK_IN', shift_assignment_id: assignment.id };
    }

    // If session exists, check if they already clocked out
    const hasClockedOut = await this.prisma.attendanceLog.findFirst({
      where: { sessionId: session.id, logType: LogType.CLOCK_OUT },
    });

    if (hasClockedOut) {
      return {
        status: 'COMPLETED',
        message: 'Tugas hari ini selesai. Terima kasih.',
      };
    }

    if (now > latestClockOut) {
      return {
        status: 'FORFEIT',
        message: 'Lupa absen pulang (Sistem ditutup).',
      };
    }

    return { status: 'CAN_CLOCK_OUT', attendance_session_id: session.id };
  }
}
