import {
  Injectable,
  BadRequestException,
  NotFoundException,
} from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';
import { CreatePatrolDto } from './dto/create-patrol.dto';

@Injectable()
export class PatrolsService {
  constructor(private prisma: PrismaService) {}

  async createPatrolReport(linmasId: string, dto: CreatePatrolDto) {
    // 1. Verify the evidence photo exists in our database from the Storage Module
    const photoFile = await this.prisma.file.findFirst({
      where: { cdn_url: dto.photo_url },
    });

    if (!photoFile) {
      throw new BadRequestException(
        'Patrol evidence photo not found. Please upload first.',
      );
    }

    // 2. Save the patrol report
    const patrol = await this.prisma.patrolReport.create({
      data: {
        linmasId: linmasId,
        patrolType: dto.patrol_type,
        description: dto.description,
        latitude: dto.latitude,
        longitude: dto.longitude,
        photoFileId: photoFile.id,
        reportedAt: new Date(dto.reported_at),
      },
    });

    return { patrol_id: patrol.id };
  }

  async getPatrols() {
    const patrols = await this.prisma.patrolReport.findMany({
      orderBy: { reportedAt: 'desc' },
      include: {
        linmas: { include: { regu: true } },
        photoFile: true,
      },
    });

    return patrols.map((p) => ({
      patrol_id: p.id,
      reporter: {
        linmas_id: p.linmas.userId,
        full_name: p.linmas.fullName,
        regu_name: p.linmas.regu?.name,
      },
      patrol_type: p.patrolType,
      description: p.description,
      location: {
        latitude: parseFloat(p.latitude.toString()),
        longitude: parseFloat(p.longitude.toString()),
      },
      photo_url: p.photoFile.cdn_url,
      reported_at: p.reportedAt.toISOString(),
    }));
  }

  async getPatrolDetail(patrolId: string) {
    const patrol = await this.prisma.patrolReport.findUnique({
      where: { id: patrolId },
      include: {
        linmas: { include: { regu: true } },
        photoFile: true,
      },
    });

    if (!patrol) throw new NotFoundException('Patrol report not found');

    return {
      patrol_id: patrol.id,
      reporter: {
        linmas_id: patrol.linmas.userId,
        full_name: patrol.linmas.fullName,
        regu_name: patrol.linmas.regu?.name || null,
      },
      patrol_type: patrol.patrolType,
      description: patrol.description,
      location: {
        latitude: parseFloat(patrol.latitude.toString()),
        longitude: parseFloat(patrol.longitude.toString()),
      },
      photo_url: patrol.photoFile.id,
      reported_at: patrol.reportedAt.toISOString(),
    };
  }
}
