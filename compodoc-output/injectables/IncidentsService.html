import {
  Injectable,
  NotFoundException,
  BadRequestException,
  ConflictException,
} from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';
import { IncidentStatus, ResponseType, Role } from '@prisma/client';
import { CreateIncidentDto } from './dto/create-incident.dto';
import { ClaimIncidentDto } from './dto/claim-incident.dto';
import { UpdateIncidentStatusDto } from './dto/update-incident-status.dto';
import { InterveneIncidentDto } from './dto/intervene-incident.dto';
import { NotificationsService } from '../notifications/notifications.service';

@Injectable()
export class IncidentsService {
  constructor(
    private prisma: PrismaService,
    private notificationsService: NotificationsService,
  ) {}

  async createIncident(wargaId: string, dto: CreateIncidentDto) {
    const photoFile = await this.prisma.file.findFirst({
      where: { cdn_url: dto.photo_url },
    });

    if (!photoFile) {
      throw new BadRequestException(
        'Evidence photo not found. Please upload first.',
      );
    }

    const incident = await this.prisma.incident.create({
      data: {
        reportedByWargaId: wargaId,
        title: `[${dto.incident_type}] ${dto.title}`,
        description: dto.description,
        latitude: dto.gps_location.latitude,
        longitude: dto.gps_location.longitude,
        photoFileId: photoFile.id,
        reportedAt: new Date(dto.reported_at),
        status: IncidentStatus.MENUNGGU,
      },
    });

    // TODO: Trigger FCM Topic "security_alerts_rw07" (Implemented in Step 10)
    await this.notificationsService.sendToTopic(
      'security_alerts_rw07',
      '🚨 DARURAT: ' + dto.incident_type,
      dto.title,
      { incident_id: incident.id },
    );

    return {
      incident_id: incident.id,
      status: incident.status,
    };
  }

  async getIncidents() {
    const incidents = await this.prisma.incident.findMany({
      orderBy: { reportedAt: 'desc' },
      include: {
        responses: {
          include: { responder: { include: { linmasProfile: true } } },
          orderBy: { createdAt: 'desc' },
          take: 1, // Get the latest handler
        },
      },
    });

    return incidents.map((inc) => {
      const activeHandler =
        inc.responses.length > 0 ? inc.responses[0].responder : null;
      return {
        incident_id: inc.id,
        title: inc.title,
        status: inc.status,
        handler: activeHandler
          ? {
              linmas_id: activeHandler.id,
              full_name: activeHandler.linmasProfile?.fullName || 'Unknown',
            }
          : null,
      };
    });
  }

  async getIncidentDetail(incidentId: string) {
    const incident = await this.prisma.incident.findUnique({
      where: { id: incidentId },
      include: {
        responses: {
          include: { responder: { include: { linmasProfile: true } } },
          orderBy: { createdAt: 'desc' },
          take: 1,
        },
      },
    });

    if (!incident) throw new NotFoundException('Incident not found');

    const activeHandler =
      incident.responses.length > 0 ? incident.responses[0].responder : null;

    return {
      incident_id: incident.id,
      status: incident.status,
      handler: activeHandler
        ? {
            linmas_id: activeHandler.id,
            full_name: activeHandler.linmasProfile?.fullName || 'Unknown',
          }
        : null,
    };
  }

  async claimIncident(incidentId: string, dto: ClaimIncidentDto) {
    return this.prisma.$transaction(async (prisma) => {
      const incident = await prisma.incident.findUnique({
        where: { id: incidentId },
      });

      if (!incident) throw new NotFoundException('Incident not found');
      if (incident.status !== IncidentStatus.MENUNGGU) {
        throw new ConflictException(
          'Incident has already been claimed or finished',
        );
      }

      await prisma.incident.update({
        where: { id: incidentId },
        data: { status: IncidentStatus.DITANGANI },
      });

      const response = await prisma.incidentResponse.create({
        data: {
          incidentId: incidentId,
          responderId: dto.linmas_id,
          responseType: ResponseType.CLAIMED,
          createdAt: new Date(dto.claimed_at),
        },
        include: {
          responder: { include: { linmasProfile: true } },
        },
      });

      const otherLinmas = await prisma.user.findMany({
        where: {
          role: Role.LINMAS,
          id: { not: dto.linmas_id },
          fcm_token: { not: null },
        },
        select: { fcm_token: true },
      });

      const tokens = otherLinmas
        .map((l) => l.fcm_token!)
        .filter((t) => t.length > 0);

      if (tokens.length > 0) {
        // Triggered via global NotificationsService
        await this.notificationsService.sendToDevices(
          tokens,
          'Incident Ditangani',
          `Insiden ${incident.title} sudah ditangani oleh ${response.responder.linmasProfile?.fullName}`,
          { incident_id: incidentId },
        );
      }
      // TODO: Trigger FCM Follow-up to others (Implemented in Step 10)

      return {
        incident_id: incidentId,
        status: IncidentStatus.DITANGANI,
        handler_name: response.responder.linmasProfile?.fullName || 'Unknown',
      };
    });
  }

  async getActiveIncidents() {
    const activeIncidents = await this.prisma.incident.findMany({
      where: {
        status: {
          in: [
            IncidentStatus.MENUNGGU,
            IncidentStatus.DITANGANI,
            IncidentStatus.DIALIHKAN,
          ],
        },
      },
      orderBy: { reportedAt: 'desc' },
      include: {
        responses: {
          include: { responder: { include: { linmasProfile: true } } },
          orderBy: { createdAt: 'desc' },
          take: 1, // Get the latest handler
        },
      },
    });

    return activeIncidents.map((inc) => {
      const activeHandler =
        inc.responses.length > 0 ? inc.responses[0].responder : null;
      return {
        incident_id: inc.id,
        title: inc.title,
        status: inc.status,
        handler: activeHandler
          ? {
              linmas_id: activeHandler.id,
              full_name: activeHandler.linmasProfile?.fullName || 'Unknown',
            }
          : null,
      };
    });
  }

  async updateIncidentStatus(incidentId: string, dto: UpdateIncidentStatusDto) {
    const incident = await this.prisma.incident.findUnique({
      where: { id: incidentId },
      include: { responses: { orderBy: { createdAt: 'desc' }, take: 1 } },
    });

    if (!incident) throw new NotFoundException('Incident not found');

    await this.prisma.$transaction(async (prisma) => {
      await prisma.incident.update({
        where: { id: incidentId },
        data: { status: dto.status },
      });

      if (incident.responses.length > 0) {
        await prisma.incidentResponse.update({
          where: { id: incident.responses[0].id },
          data: { notes: dto.resolution_notes },
        });
      }
    });

    return true;
  }

  async interveneIncident(
    incidentId: string,
    koordinatorId: string,
    dto: InterveneIncidentDto,
  ) {
    const incident = await this.prisma.incident.findUnique({
      where: { id: incidentId },
    });
    if (!incident) throw new NotFoundException('Incident not found');

    await this.prisma.$transaction(async (prisma) => {
      await prisma.incident.update({
        where: { id: incidentId },
        data: { status: IncidentStatus.DITANGANI }, // Lock it back to handled
      });

      await prisma.incidentResponse.create({
        data: {
          incidentId: incidentId,
          responderId: dto.new_handler_linmas_id,
          responseType: ResponseType.INTERVENED,
          notes: `[Intervensi Koordinator]: ${dto.intervention_reason}`,
        },
      });
    });

    return true;
  }

  async getMyIncidents(wargaId: string) {
    const incidents = await this.prisma.incident.findMany({
      where: { reportedByWargaId: wargaId },
      orderBy: { reportedAt: 'desc' },
      select: {
        id: true,
        status: true,
        photoFileId: true,
      },
    });

    return incidents.map((inc) => ({
      incident_id: inc.id,
      status: inc.status,
      photoFileId: inc.photoFileId,
    }));
  }
}
